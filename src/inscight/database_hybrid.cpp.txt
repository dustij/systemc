#define SQL_ERROR(...)                \
    do                                \
    {                                 \
        fprintf(stderr, __VA_ARGS__); \
        fprintf(stderr, "\n");        \
        fflush(stderr);               \
        abort();                      \
    } while (0)

#define SQL_ERROR_ON(ret, ...)      \
    do                              \
    {                               \
        if (ret != SQLITE_OK)       \
        {                           \
            SQL_ERROR(__VA_ARGS__); \
        }                           \
    } while (0)

    database_hybrid::db::db() : m_db()
    {
    }

    database_hybrid::db::~db()
    {
        if (m_db != nullptr)
            sqlite3_close(m_db);
    }

    void database_hybrid::db::open()
    {
        static char name[256];
        memset(name, 0, sizeof(name));
        snprintf(name, sizeof(name), "sim.%d.db", (int)getpid());
        int ret = sqlite3_open(name, &m_db);
        SQL_ERROR_ON(ret, "failed to open database: %s", sqlite3_errmsg(m_db));
    }

    database_hybrid::stmt::stmt(db &d, const std::string &text) : m_db(d),
                                                                  m_stmt(nullptr),
                                                                  m_text(text)
    {
    }

    database_hybrid::stmt::~stmt()
    {
        if (m_stmt)
            sqlite3_finalize(m_stmt);
    }

    void database_hybrid::stmt::prepare()
    {
        int ret = sqlite3_prepare_v2(m_db.handle(), text(), -1, &m_stmt, nullptr);
        SQL_ERROR_ON(ret, "failed to prepare '%s': %s", text(), m_db.error());
    }

    void database_hybrid::stmt::bind(size_t idx, unsigned long long val)
    {
        if (m_stmt == nullptr)
            prepare();

        int ret = sqlite3_bind_int64(m_stmt, idx, val);
        SQL_ERROR_ON(ret, "failed to bind arg %zu: %s", idx, m_db.error());
    }

    void database_hybrid::stmt::bind(size_t idx, const char *str)
    {
        if (m_stmt == nullptr)
            prepare();

        int ret = sqlite3_bind_text(m_stmt, idx, str, -1, SQLITE_TRANSIENT);
        SQL_ERROR_ON(ret, "failed to bind arg %zu: %s", idx, m_db.error());
    }

    void database_hybrid::stmt::execute()
    {
        if (m_stmt == nullptr)
            prepare();

        int ret = sqlite3_step(m_stmt);
        if (ret != SQLITE_DONE)
            SQL_ERROR("failed to evaluate '%s': %s", text(), m_db.error());

        ret = sqlite3_clear_bindings(m_stmt);
        SQL_ERROR_ON(ret, "failed to clear command bindings");
        ret = sqlite3_reset(m_stmt);
        SQL_ERROR_ON(ret, "failed to reset command");
    }

    void database_hybrid::exec(const std::string &cmd)
    {
        stmt x(m_db, cmd);
        x.execute();
    }

    void database_hybrid::init()
    {
        struct stat info;

        m_db.open();
        exec("CREATE TABLE meta(pid BIGINT PRIMARY KEY, path STRING, user STRING, version STRING, time DATETIME);");
        exec("CREATE TABLE transactions(id INTEGER PRIMARY KEY, st BIGINT, dir INTEGER, port BIGINT, proto INTEGER);");

        if (stat(OUTPUT_DIR.c_str(), &info) != 0)
        {
            // Directory does not exist, create it
            if (mkdir(OUTPUT_DIR.c_str(), 0777) == 0)
            {
                std::cout << "Created directory: " << OUTPUT_DIR << std::endl;
            }
            else
            {
                std::cerr << "Error: Unable to create directory " << OUTPUT_DIR << std::endl;
            }
        }
    }

    void database_hybrid::begin(size_t n)
    {
        m_stmt_tx_begin.execute();
    }

    void database_hybrid::end(size_t n)
    {
        m_stmt_tx_end.execute();
    }

    void database_hybrid::write_to_parquet(const std::vector<std::tuple<uint64_t, uint64_t, int, uint64_t, std::string>> &transactions)
    {
        // std::lock_guard<std::mutex> lock(m_parquet_mutex);
        m_parquet_mutex.lock();
        std::cout << "[Thread " << std::this_thread::get_id() << "] Entering write_to_parquet()" << ++yoda << std::endl
                  << std::flush;

        if (transactions.empty())
        {
            std::cerr << "Warning: Empty transaction batch passed to Parquet writer." << std::endl;
        }

        if (!m_memory_pool)
        {
            std::cerr << "Fatal Error: Memory pool already destroyed." << std::endl
                      << std::flush;
            std::exit(1);
        }

        // Create Arrow Builders for columns
        arrow::UInt64Builder obj_builder(m_memory_pool), timestamp_builder(m_memory_pool), proto_builder(m_memory_pool);
        arrow::Int64Builder dir_builder;
        arrow::StringBuilder json_builder;

        void(obj_builder.Resize(PARQUET_BATCH_SIZE));
        void(timestamp_builder.Resize(PARQUET_BATCH_SIZE));
        void(dir_builder.Resize(PARQUET_BATCH_SIZE));
        void(proto_builder.Resize(PARQUET_BATCH_SIZE));
        void(json_builder.Resize(PARQUET_BATCH_SIZE));

        // Append data to Arrow builders
        for (const auto &[obj, timestamp, direction, proto, json] : transactions)
        {
            auto res = obj_builder.Append(obj);
            if (!res.ok())
            {
                std::cout << "Arrow builder obj append failed!!" << std::endl
                          << std::flush;
            }
            res = timestamp_builder.Append(timestamp);
            if (!res.ok())
            {
                std::cout << "Arrow builder timestamp append failed!!" << std::endl
                          << std::flush;
            }
            res = dir_builder.Append(direction);
            if (!res.ok())
            {
                std::cout << "Arrow builder direction append failed!!" << std::endl
                          << std::flush;
            }
            res = proto_builder.Append(proto);
            if (!res.ok())
            {
                std::cout << "Arrow builder proto append failed!!" << std::endl
                          << std::flush;
            }
            res = json_builder.Append(json);
            if (!res.ok())
            {
                std::cout << "Arrow builder json  append failed!!" << std::endl
                          << std::flush;
            }
        }

        // Create Arrow arrays
        std::shared_ptr<arrow::Array> obj_array, timestamp_array, dir_array, proto_array, json_array;

        auto res = obj_builder.Finish(&obj_array);
        if (!res.ok())
        {
            std::cout << "Arrow builder obj array append failed!!" << std::endl
                      << std::flush;
        }

        res = timestamp_builder.Finish(&timestamp_array);
        if (!res.ok())
        {
            std::cout << "Arrow builder timestamp array append failed!!" << std::endl
                      << std::flush;
        }

        res = dir_builder.Finish(&dir_array);
        if (!res.ok())
        {
            std::cout << "Arrow builder dir array  append failed!!" << std::endl
                      << std::flush;
        }

        res = proto_builder.Finish(&proto_array);
        if (!res.ok())
        {
            std::cout << "Arrow builder proto array  append failed!!" << std::endl
                      << std::flush;
        }

        res = json_builder.Finish(&json_array);
        if (!res.ok())
        {
            std::cout << "Arrow builder json array append failed!!" << std::endl
                      << std::flush;
        }

        // Define schema
        auto schema = arrow::schema({arrow::field("obj", arrow::uint64()),
                                     arrow::field("timestamp", arrow::uint64()),
                                     arrow::field("direction", arrow::int64()),
                                     arrow::field("protocol", arrow::uint64()),
                                     arrow::field("payload", arrow::utf8())});

        // Create table
        auto table = arrow::Table::Make(schema, {obj_array, timestamp_array, dir_array, proto_array, json_array});

        // 🔹 Generate a unique filename for each batch
        m_batch_counter++; // Increment batch number

        // Write to Parquet file with batch numbering
        std::string file_name = OUTPUT_DIR + "batch_" + std::to_string(m_batch_counter) + ".parquet";
        std::shared_ptr<arrow::io::FileOutputStream> outfile;
        PARQUET_ASSIGN_OR_THROW(outfile, arrow::io::FileOutputStream::Open(file_name));
        int rgs = 64 * 1024;
        PARQUET_THROW_NOT_OK(parquet::arrow::WriteTable(*table, m_memory_pool, outfile, rgs));
        parquet_buffer.clear();
        m_parquet_mutex.unlock();
        // std::cout << "Written " << transactions.size() << " transactions to " << file_name << std::endl;

        std::cout << "[Thread " << std::this_thread::get_id() << "] exiting write_to_parquet() " << yoda << std::endl
                  << std::flush;
    }

    void database_hybrid::process_transaction(uint64_t obj, uint64_t timestamp, int direction, uint64_t proto, const std::string &json)
    {
        parquet_buffer.emplace_back(obj, timestamp, direction, proto, json);

        if (parquet_buffer.size() >= PARQUET_BATCH_SIZE)
        {
            write_to_parquet(parquet_buffer);
        }
    }

    void database_hybrid::flush_parquet_buffer()
    {
        if (!parquet_buffer.empty())
        {
            std::cout << "flush_parquet_buffer called by thread " << std::this_thread::get_id() << std::endl
                      << std::flush;
            std::cout << "Trying to write " << parquet_buffer.size() << " transactions" << std::endl
                      << std::flush;
            write_to_parquet(parquet_buffer);
        }
    }

    void database_hybrid::transaction_trace_fw(id_t obj, sysc_time_t st, protocol_kind proto, const char *json)
    {
        m_stmt_insert_transaction.bind(1, st);
        m_stmt_insert_transaction.bind(2, obj);
        m_stmt_insert_transaction.bind(3, 0ull);  // Direction (fw = 0)
        m_stmt_insert_transaction.bind(4, proto); // Fix: Bind only 4 values
        m_stmt_insert_transaction.execute();

        // Send to Parquet buffer
        process_transaction(obj, st, 0, proto, json);
    }

    void database_hybrid::transaction_trace_bw(id_t obj, sysc_time_t st, protocol_kind proto, const char *json)
    {
        m_stmt_insert_transaction.bind(1, st);
        m_stmt_insert_transaction.bind(2, obj);
        m_stmt_insert_transaction.bind(3, 1ull);
        m_stmt_insert_transaction.bind(4, proto);

        // We won't write the json anymore
        // m_stmt_insert_transaction.bind(5, json);
        m_stmt_insert_transaction.execute();

        process_transaction(obj, st, 1, proto, json);
    }

    {

        m_batch_counter = 1;
    }
